function importExcelCustomFields(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = (e) => { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); const firstSheetName = workbook.SheetNames[0]; const sheet = workbook.Sheets[firstSheetName]; const json = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: "" }); if (!json || json.length < 2) { alert('ไฟล์ Excel ไม่มีข้อมูล หรือโครงสร้างไม่ถูกต้อง (ต้องมีหัวตารางที่บรรทัดแรก)'); return; } // ดึงหัวตารางจากบรรทัดแรก (Row 0) const rawHeaders = json[0]; const dynamicFields = []; rawHeaders.forEach(h => { const cleanH = String(h || '').trim(); if (cleanH && !cleanH.includes('ลำดับ') && cleanH.toLowerCase() !== 'no' && cleanH.toLowerCase() !== 'no.') { dynamicFields.push(cleanH); } }); // กำหนดฟิลด์หัวข้อตาราง currentFields = dynamicFields.length > 0 ? dynamicFields : ["ชื่อ-นามสกุล", "อีเมล"]; if (!eventsData[currentEventId]) eventsData[currentEventId] = {}; eventsData[currentEventId].fields = currentFields; // แปลงข้อมูลแถวต่างๆ เป็นวัตถุผู้ลงทะเบียน const importedAttendees = {}; let importCount = 0; for (let i = 1; i < json.length; i++) { const row = json[i]; if (!row || row.every(cell => cell === "")) continue; // ข้ามแถวว่าง const attObj = {}; rawHeaders.forEach((h, colIdx) => { const cleanH = String(h || '').trim(); if (cleanH) { attObj[cleanH] = row[colIdx] !== undefined ? String(row[colIdx]).trim() : ''; } }); // สร้าง ID อัตโนมัติหากไม่มีการระบุ id/รหัส const uniqueId = attObj['id'] || attObj['รหัส'] || attObj['empId'] || `ID_${Date.now()}_${i}`; attObj['_id'] = uniqueId; importedAttendees[uniqueId] = attObj; importCount++; } if (importCount === 0) { alert('ไม่พบแถวข้อมูลในไฟล์ Excel'); return; } // บันทึกลง Firebase Database DB_ADAPTOR.set(`attendees_${currentEventId}`, importedAttendees, (err) => { if (err) { alert('เกิดข้อผิดพลาดในการบันทึกข้อมูลลงฐานข้อมูล'); console.error(err); } else { DB_ADAPTOR.set('events', eventsData); alert(`นำเข้าข้อมูลสำเร็จทั้งหมด ${importCount} รายการ`); renderAttendeesView(); // บังคับรีเฟรชหน้าจอทันที } }); } catch (err) { alert('ไม่สามารถอ่านไฟล์ Excel ได้ กรุณาตรวจสอบว่าเป็นไฟล์ .xlsx หรือ .xls ที่ถูกต้อง'); console.error(err); } }; reader.readAsArrayBuffer(file); }
